//// twoSum
function twoSum(nums, target) { const seen = new Map(); for (let i = 0; i < nums.length; i++) { const j = seen.get(target - nums[i]); if (j !== undefined) return [j, i]; seen.set(nums[i], i); } return null; }
//// isValid
function isValid(s) { const st = []; for (const c of s) { if ('([{'.includes(c)) st.push(c); else { const o = st.pop(); if (!o || ('([{'.indexOf(o) !== ')]}'.indexOf(c))) return false; } } return st.length === 0; }
//// reverseString
function reverseString(s) { return s.split('').reverse().join(''); }
//// isPalindrome
function isPalindrome(s) { const t = s.toLowerCase().replace(/[^a-z0-9]/g, ''); return t === t.split('').reverse().join(''); }
//// fizzBuzz
function fizzBuzz(n) { const out = []; for (let i = 1; i <= n; i++) out.push(i % 15 === 0 ? 'FizzBuzz' : i % 3 === 0 ? 'Fizz' : i % 5 === 0 ? 'Buzz' : String(i)); return out; }
//// findMax
function findMax(nums) { return Math.max(...nums); }
//// digitSum
function digitSum(n) { let s = 0; while (n > 0) { s += n % 10; n = Math.floor(n / 10); } return s; }
//// factorial
function factorial(n) { let r = 1; for (let i = 2; i <= n; i++) r *= i; return r; }
//// fib
function fib(n) { let a = 0, b = 1; for (let i = 0; i < n; i++) { [a, b] = [b, a + b]; } return a; }
//// isPrime
function isPrime(n) { if (n < 2) return false; for (let i = 2; i * i <= n; i++) if (n % i === 0) return false; return true; }
//// containsDuplicate
function containsDuplicate(nums) { return new Set(nums).size !== nums.length; }
//// isAnagram
function isAnagram(s, t) { return s.length === t.length && s.split('').sort().join('') === t.split('').sort().join(''); }
//// moveZeroes
function moveZeroes(nums) { const nz = nums.filter(x => x !== 0); return nz.concat(new Array(nums.length - nz.length).fill(0)); }
//// maxProfit
function maxProfit(prices) { let min = Infinity, best = 0; for (const p of prices) { min = Math.min(min, p); best = Math.max(best, p - min); } return best; }
//// climbStairs
function climbStairs(n) { let a = 1, b = 1; for (let i = 1; i < n; i++) { [a, b] = [b, a + b]; } return b; }
//// search
function search(nums, target) { let lo = 0, hi = nums.length - 1; while (lo <= hi) { const mid = (lo + hi) >> 1; if (nums[mid] === target) return mid; if (nums[mid] < target) lo = mid + 1; else hi = mid - 1; } return -1; }
//// isPowerOfTwo
function isPowerOfTwo(n) { return n > 0 && (n & (n - 1)) === 0; }
//// singleNumber
function singleNumber(nums) { return nums.reduce((a, b) => a ^ b, 0); }
//// missingNumber
function missingNumber(nums) { let r = nums.length; nums.forEach((x, i) => { r ^= i ^ x; }); return r; }
//// maxSubArray
function maxSubArray(nums) { let cur = nums[0], best = nums[0]; for (let i = 1; i < nums.length; i++) { cur = Math.max(nums[i], cur + nums[i]); best = Math.max(best, cur); } return best; }
//// merge
function merge(intervals) { const a = intervals.map(x => x.slice()).sort((x, y) => x[0] - y[0]); const out = []; for (const iv of a) { if (!out.length || out[out.length - 1][1] < iv[0]) out.push(iv); else out[out.length - 1][1] = Math.max(out[out.length - 1][1], iv[1]); } return out; }
//// productExceptSelf
function productExceptSelf(nums) { const n = nums.length, out = new Array(n).fill(1); for (let i = 1; i < n; i++) out[i] = out[i - 1] * nums[i - 1]; let right = 1; for (let i = n - 1; i >= 0; i--) { out[i] *= right; right *= nums[i]; } return out; }
//// longestCommonPrefix
function longestCommonPrefix(strs) { let p = strs[0]; for (const s of strs) while (!s.startsWith(p)) p = p.slice(0, -1); return p; }
//// romanToInt
function romanToInt(s) { const v = { I: 1, V: 5, X: 10, L: 50, C: 100, D: 500, M: 1000 }; let t = 0; for (let i = 0; i < s.length; i++) { const c = v[s[i]], n = v[s[i + 1]] || 0; t += c < n ? -c : c; } return t; }
//// majorityElement
function majorityElement(nums) { let c = 0, count = 0; for (const x of nums) { if (count === 0) c = x; count += x === c ? 1 : -1; } return c; }
//// coinChange
function coinChange(coins, amount) { const dp = new Array(amount + 1).fill(amount + 1); dp[0] = 0; for (let a = 1; a <= amount; a++) for (const c of coins) if (c <= a) dp[a] = Math.min(dp[a], dp[a - c] + 1); return dp[amount] > amount ? -1 : dp[amount]; }
//// rob
function rob(nums) { let skip = 0, take = 0; for (const x of nums) { const next = Math.max(take, skip + x); skip = take; take = next; } return take; }
//// rotate
function rotate(nums, k) { const n = nums.length, out = new Array(n); for (let i = 0; i < n; i++) out[(i + k) % n] = nums[i]; return out; }
//// countVowels
function countVowels(s) { return (s.toLowerCase().match(/[aeiou]/g) || []).length; }
//// removeDuplicates
function removeDuplicates(nums) { if (!nums.length) return 0; let w = 1; for (let r = 1; r < nums.length; r++) if (nums[r] !== nums[w - 1]) nums[w++] = nums[r]; return w; }
